fn repository_revision(root: &Path) -> Result<String, RegistryError> {
    let head_path = root.join(".git/HEAD");
    let head = fs::read_to_string(&head_path)
        .map_err(|e| invalid(&head_path, format!("read current Git revision: {e}")))?;
    let value = head.trim();
    let revision = if let Some(reference) = value.strip_prefix("ref: ") {
        if !reference.starts_with("refs/heads/")
            || reference
                .split('/')
                .any(|part| part == ".." || part.is_empty())
        {
            return Err(invalid(root, "Git HEAD is not a local branch"));
        }
        let reference_path = root.join(".git").join(reference);
        match fs::read_to_string(&reference_path) {
            Ok(revision) => revision.trim().to_owned(),
            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
                let packed = fs::read_to_string(root.join(".git/packed-refs"))
                    .map_err(|e| invalid(root, format!("read packed Git refs: {e}")))?;

                    .lines()
                    .find_map(|line| {
                        let mut fields = line.split_whitespace();
                        let revision = fields.next()?;
                        let name = fields.next()?;
                        (name == reference).then(|| revision.to_owned())
                    })
                    .ok_or_else(|| invalid(root, "current Git branch has no commit"))?
            }
            Err(error) => return Err(invalid(&reference_path, error)),
        }
    } else {
        value.to_owned()
    };
    if revision.len() != 40
        || !revision
            .bytes()
            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
    {
        return Err(invalid(
            root,
            "current Git revision must be 40 lowercase hex",
        ));
    }
    Ok(revision)

            source_revision: self.revisions[i].clone(),
            source_root: self.roots[i].to_string_lossy().into_owned(),
            source_relative_path: skill
                .path
                .strip_prefix(&self.roots[i])
                .unwrap()
                .to_string_lossy()
                .into_owned(),
            module_type: kind.into(),
            module_subject: subject.into(),
            declared_placement: declared_placement(&skill.body).into(),
            source_sha256: sha256(skill.body.as_bytes()),
            body_sha256: sha256(text.as_bytes()),
            string: text.to_owned(),
            string_vector: skill.dependencies,
        })
    }
    fn visit(
        &self,
        id: &str,
        visiting: &mut BTreeSet<String>,
        seen: &mut BTreeSet<String>,
        output: &mut Vec<ResolvedSkill>,
    ) -> Result<(), RegistryError> {
        if seen.contains(id) {
            return Ok(());
        }
        if !visiting.insert(id.into()) {
            return Err(RegistryError::Cycle(id.into()));
        }
        let source = self
            .entries
            .get(id)
            .ok_or_else(|| RegistryError::Missing(id.into()))?;
        for dependency in &source.dependencies {
            self.visit(dependency, visiting, seen, output)?;
        }
        visiting.remove(id);
        seen.insert(id.into());
        output.push(source.clone());
        Ok(())
    }
}

        fn directory(self) -> &'static str {
            match self {

                Self::Codex => ".agents/skills",

                Self::Pi => ".pi/skills",
                Self::OpenCode => ".opencode/skills",

        fn render(self, skill: &ResolvedSkill) -> Result<String, RegistryError> {
            let mut rendered = String::new();
            let mut branches = Vec::<(bool, bool)>::new();
            let mut raw = false;

            for (index, line) in skill.body.split_inclusive('\n').enumerate() {
                let line_number = index + 1;
                let directive = line.trim();
                if raw {
                    if directive == "{% endraw %}" {
                        raw = false;
                    } else if branches.iter().all(|(selected, _)| *selected) {
                        rendered.push_str(line);
                    }
                    continue;
                }

                let selected = match directive {
                    "{% if claude %}" => Some(self == Self::Claude),
                    "{% if codex %}" => Some(matches!(self, Self::Codex | Self::CodexAlt)),
                    "{% if opencode %}" => Some(self == Self::OpenCode),
                    "{% if pi %}" => Some(self == Self::Pi),
                    "{% raw %}" => {
                        raw = true;
                        continue;
                    }
                    "{% endraw %}" => {
                        return Err(invalid(
                            &skill.path,
                            format!("line {line_number}: endraw without raw"),
                        ));
                    }
                    "{% else %}" => {
                        let Some((selected, saw_else)) = branches.last_mut() else {
                            return Err(invalid(
                                &skill.path,
                                format!("line {line_number}: else without if"),
                            ));
                        };
                        if *saw_else {
                            return Err(invalid(
                                &skill.path,
                                format!("line {line_number}: repeated else"),
                            ));
                        }
                        *selected = !*selected;
                        *saw_else = true;
                        continue;
                    }
                    "{% endif %}" => {
                        if branches.pop().is_none() {
                            return Err(invalid(
                                &skill.path,
                                format!("line {line_number}: endif without if"),
                            ));
                        }
                        continue;
                    }
                    _ => None,
                };
                if let Some(selected) = selected {
                    branches.push((selected, false));
                    continue;
                }
                if branches.iter().all(|(selected, _)| *selected) {
                    if self == Self::Claude && directive == "user-only: true" {
                        rendered.push_str("disable-model-invocation: true\n");
                    } else {
                        rendered.push_str(line);
                    }
                }
            }

            if !branches.is_empty() || raw {
                return Err(invalid(
                    &skill.path,
                    "unclosed target conditional or raw block",
                ));
            }
            if self == Self::OpenCode {
                let name = format!("name: {}\n", skill.name);
                rendered = match rendered.strip_prefix("---\n") {
                    Some(rest) => format!("---\n{name}{rest}"),
                    None => format!("---\n{name}---\n{rendered}"),
                };
            }
            Ok(rendered)
        }
    }

        fn remove_stale(&self, target: Target, current: &[String]) -> Result<(), RegistryError> {
            // A prior inventory is the ownership receipt. Without one, existing
            // directories may belong to the user and are left untouched.
            let inventory = self.safe(Path::new("modules"))?;
            let previous = match fs::read_to_string(&inventory) {
                Ok(text) => text
                    .lines()
                    .filter_map(|line| line.split_whitespace().next().map(str::to_owned))
                    .collect::<Vec<_>>(),
                Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(()),
                Err(error) => return Err(invalid(&inventory, error)),
            };
            let surface = self.safe(Path::new(target.directory()))?;
            let entries = match fs::read_dir(&surface) {
                Ok(entries) => entries,
                Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(()),
                Err(e) => return Err(invalid(&surface, e)),
            };
            for entry in entries {
                let entry = entry.map_err(|e| invalid(&surface, e))?;
                let name = entry.file_name().to_string_lossy().into_owned();
                let path = entry.path();
                let metadata = fs::symlink_metadata(&path).map_err(|e| invalid(&path, e))?;
                if previous.iter().any(|id| id == &name)
                    && !current.iter().any(|id| id == &name)
                    && !metadata.file_type().is_symlink()
                    && metadata.is_dir()
                {
                    fs::remove_dir_all(&path).map_err(|e| invalid(&path, e))?;
                }
            }
            Ok(())
        }

        fn safe(&self, relative: &Path) -> Result<PathBuf, RegistryError> {
            if relative.is_absolute()
                || relative
                    .components()
                    .any(|component| !matches!(component, Component::Normal(_)))
            {
                return Err(invalid(relative, "workspace output escapes its root"));
            }
            let root = fs::canonicalize(&self.root).map_err(|e| invalid(&self.root, e))?;
            let components = relative.components().collect::<Vec<_>>();
            let mut path = root;
            for (index, component) in components.iter().enumerate() {
                path.push(component.as_os_str());
                match fs::symlink_metadata(&path) {
                    Ok(metadata) if metadata.file_type().is_symlink() => {
                        return Err(invalid(&path, "workspace output crosses a symlink"));
                    }
                    Ok(metadata) if index + 1 < components.len() && !metadata.is_dir() => {
                        return Err(invalid(&path, "workspace output parent is not a directory"));
                    }
                    Ok(_) => (),
                    Err(error) if error.kind() == std::io::ErrorKind::NotFound => (),
                    Err(error) => return Err(invalid(&path, error)),
                }
            }
            Ok(path)
        }
    }

    fn write_atomic(path: &Path, contents: String) -> Result<(), RegistryError> {
        let parent = path.parent().expect("workspace output parent");
        let mut temporary = None;
        for _ in 0..16 {
            let serial = NEXT_TEMP.fetch_add(1, Ordering::Relaxed);
            let candidate = parent.join(format!(".curriculum-{}-{serial}.tmp", std::process::id()));
            match fs::OpenOptions::new()
                .write(true)
                .create_new(true)
                .open(&candidate)
            {
                Ok(mut file) => {
                    if let Err(error) = file.write_all(contents.as_bytes()) {
                        let _ = fs::remove_file(&candidate);
                        return Err(invalid(&candidate, error));
                    }
                    if let Err(error) = fs::rename(&candidate, path) {
                        let _ = fs::remove_file(&candidate);
                        return Err(invalid(path, error));
                    }
                    temporary = Some(());
                    break;
                }
                Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => continue,
                Err(error) => return Err(invalid(&candidate, error)),
            }
        }
        if temporary.is_none() {
            return Err(invalid(path, "cannot allocate an atomic output file"));
        }
        Ok(())
    }

    fn user_only(body: &str) -> bool {
        body.strip_prefix("---\n")
            .and_then(|body| body.split_once("\n---\n"))
            .is_some_and(|(frontmatter, _)| {
                frontmatter
                    .lines()
                    .any(|line| line.trim() == "user-only: true")
            })
    }
}
fn dependencies(path: &Path, body: &str) -> Result<Vec<String>, RegistryError> {
    let Some((header, _)) = body
        .strip_prefix("---\n")
        .and_then(|b| b.split_once("\n---\n"))
    else {
        return Ok(vec![]);
    };
    let Some(value) = header.lines().find_map(|l| l.strip_prefix("dependencies:")) else {
        return Ok(vec![]);
    };
    let value = value.trim();
    let Some(inner) = value.strip_prefix('[').and_then(|v| v.strip_suffix(']')) else {
        return Err(invalid(path, "dependencies must be an inline ID list"));
    };
    let ids: Vec<String> = inner
        .split(|c: char| c == ',' || c.is_whitespace())
        .filter(|s| !s.is_empty())
        .map(dependency_id)
        .collect();
    let mut seen = BTreeSet::new();
    if ids.iter().any(|id| !valid_id(id) || !seen.insert(id)) {
        return Err(invalid(path, "invalid or duplicate dependency ID"));
    }
    Ok(ids)
}
/// A dependency names a module as `type:subject` or as `type-subject`; both
/// are the same ID, held as `type:subject`. No module type has a hyphen, so
/// the first hyphen of a hyphen form separates its type from its subject.
fn dependency_id(word: &str) -> String {
    match word.split_once('-') {
        Some((kind, subject)) if !word.contains(':') && MODULE_TYPES.contains(&kind) => {
            format!("{kind}:{subject}")
        }
        _ => word.to_owned(),
    }
}

            source_sha256: module.source_sha256,
            body_sha256: module.body_sha256,
            rendered_sha256,
            startup_placement: if is_system {

            } else {
                StartupPlacement::User
            },
            integer: ordinal,
        });
    }

    let user_text = render(&user);

        records_payload_sha256: sha256(records_text.as_bytes()),

fn records(root: &Path, flows: &[String]) -> Result<String, String> {

        {
            return Err(format!("record missing: {flow}"));
        }
        let flow_root =
            std::fs::canonicalize(&flow_root).map_err(|_| format!("record missing: {flow}"))?;

            return Err(format!("record missing: {flow}"));
        }
        let mut paths = Vec::new();

            let directory = flow_root.join(topic);
            match std::fs::symlink_metadata(&directory) {
                Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
                Err(_) => return Err(format!("record missing: {flow}")),
                Ok(metadata) if !metadata.is_dir() || metadata.file_type().is_symlink() => {
                    return Err(format!("record missing: {flow}"));
                }
                Ok(_) => {}
            }
            let mut topic_paths = std::fs::read_dir(&directory)
                .map_err(|_| format!("record missing: {flow}"))?
                .map(|entry| entry.map(|entry| entry.path()))
                .collect::<Result<Vec<_>, _>>()
                .map_err(|_| format!("record missing: {flow}"))?;
            topic_paths.retain(|path| {
                path.extension().is_some_and(|extension| extension == "md")
                    && path
                        .file_name()
                        .and_then(|name| name.to_str())
                        .is_some_and(|name| !name.starts_with("archive-"))
            });
            topic_paths.sort();
            paths.extend(topic_paths);
        }
        for path in paths {
            let metadata =
                std::fs::symlink_metadata(&path).map_err(|_| format!("record missing: {flow}"))?;
            if !metadata.is_file() || metadata.file_type().is_symlink() {
                return Err(format!("record missing: {flow}"));
            }
            let bytes = std::fs::read(&path).map_err(|_| format!("record missing: {flow}"))?;
            if std::fs::read(&path).ok().as_deref() != Some(bytes.as_slice()) {
                return Err(format!("record missing: {flow}"));
            }

            if !output.is_empty() && !output.ends_with('\n') {
                output.push('\n');
            }
            output.push_str(&text);
            if !output.ends_with('\n') {
                output.push('\n');
            }
        }
    }
    Ok(output)
}

fn render(modules: &[(String, String)]) -> String {
    if modules.is_empty() {
        return String::new();
    }
    let mut text = format!(
        "These are your skills, given here; do not reload them.\n\nModule IDs: {}\n",
        modules
            .iter()
            .map(|(id, _)| id.as_str())
            .collect::<Vec<_>>()
            .join(", ")
    );
    for (id, body) in modules {
        text.push_str(&format!(
            "\n<!-- module: {id} -->\n{body}\n<!-- end-module: {id} -->\n"
        ));
    }
    text
}
fn identifier(value: &str) -> bool {
    value.split_once(':').is_some_and(|(kind, subject)| {
        matches!(
            kind,
            "spirit"
                | "intent"
                | "vision"
                | "knowledge"
                | "inquiry"
                | "compensation"
                | "role"
                | "raw"
        ) && !subject.is_empty()
            && !subject.contains(':')
            && subject
                .bytes()
                .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
    })
}
fn system_module(value: &str) -> bool {
    value.starts_with("spirit:") || value.starts_with("intent:")
}
fn flow_id(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|byte| byte.is_ascii_alphanumeric() || byte == b'-')
}
fn word(value: &str) -> bool {
    !value.is_empty()
        && value
            .bytes()
            .all(|b| b.is_ascii_alphanumeric() || b == b'-')
}

/// One big-endian u32 length-prefixed rkyv `Query` frame out, one
/// `Outcome` frame back.
impl ExchangesCurriculumFrames for CurriculumComposer {
    fn exchange<N: CurriculumQuerying>(&self, input: N) -> Result<CurriculumOutcome, String> {
        let request = rkyv::to_bytes::<rkyv::rancor::Error>(&input.query())
            .map_err(|error| format!("query not archived: {error}"))?;
        let length = u32::try_from(request.len()).map_err(|_| "query frame too large")?;
        let mut stream = UnixStream::connect(&self.socket)
            .map_err(|error| format!("Curriculum socket unreachable: {error}"))?;
        stream
            .set_read_timeout(Some(EXCHANGE_TIMEOUT))
            .and_then(|_| stream.set_write_timeout(Some(EXCHANGE_TIMEOUT)))
            .map_err(|error| error.to_string())?;
        stream
            .write_all(&length.to_be_bytes())
            .and_then(|_| stream.write_all(&request))
            .and_then(|_| stream.flush())
            .map_err(|error| format!("query not written: {error}"))?;
        let mut prefix = [0_u8; 4];
        stream
            .read_exact(&mut prefix)
            .map_err(|error| format!("answer not read: {error}"))?;
        let length = u32::from_be_bytes(prefix) as usize;
        if length > ANSWER_FRAME_LIMIT {
            return Err("answer frame exceeds its limit".into());
        }
        let mut frame = vec![0_u8; length];
        stream
            .read_exact(&mut frame)
            .map_err(|error| format!("answer not read: {error}"))?;
        rkyv::from_bytes::<CurriculumOutcome, rkyv::rancor::Error>(&frame)
            .map_err(|error| format!("answer unreadable: {error}"))
    }
}

impl ComposesFlowContext for CurriculumComposer {
    fn compose_flow_context<N: ContextRequesting>(

/// Cuts a rendered text back into its modules, checking each module's slice
/// against its rendered SHA-256 and the whole against the rendering.
impl CutsRenderedText for str {
    fn rendered_segments<N: ProvenanceListing>(&self, input: N) -> Option<Vec<String>> {
        let rows = input.provenance();
        if rows.is_empty() {
            return self.is_empty().then(Vec::new);
        }
        let opening = format!(
            "{}\n\nModule IDs: {}\n",
            CONTEXT_OPENING,
            rows.iter()
                .map(|row| row.module_identity.as_str())
                .collect::<Vec<_>>()
                .join(", ")
        );
        let mut rest = self.strip_prefix(&opening)?;
        let mut segments = Vec::new();
        for (index, row) in rows.iter().enumerate() {
            let identity = &row.module_identity;
            let open = format!("\n<!-- module: {identity} -->\n");
            let close = format!("\n<!-- end-module: {identity} -->\n");
            let after_open = rest.strip_prefix(&open)?;
            let body_end = match rows.get(index + 1) {
                Some(next) => {
                    let next_open = format!("{close}\n<!-- module: {} -->\n", next.module_identity);
                    after_open.find(&next_open)?
                }
                None => after_open.strip_suffix(&close)?.len(),
            };
            let body = &after_open[..body_end];
            if format!("{:x}", Sha256::digest(body.as_bytes())) != row.rendered_sha256 {
                return None;
            }
            let segment = format!("{open}{body}{close}");
            rest = &rest[segment.len()..];
            segments.push(segment);
        }
        if !rest.is_empty() {
            return None;
        }
        segments[0].insert_str(0, &opening);
        Some(segments)
    }
}

impl VerifiesContext for VerifiedContext {
    fn verified<N: ContextRequesting, O: StartupComposing>(
        input_0: N,
        input_1: O,
    ) -> Result<Self, ContextRefusal> {
        let request = input_0.context_request();
        let composition = input_1.composition();
        let digest = |text: &str| format!("{:x}", Sha256::digest(text.as_bytes()));
        if digest(&composition.system_text) != composition.system_payload_sha256
            || digest(&composition.user_text) != composition.user_payload_sha256
            || digest(&composition.records_text) != composition.records_payload_sha256
        {
            return Err(ContextRefusal::PayloadMismatch);
        }
        // Records are composed only for a launch that remembers flows.
        if request.remembered_flows.is_empty() && !composition.records_text.is_empty() {
            return Err(ContextRefusal::PayloadMismatch);
        }
        let rows = &composition.context_provenance_vector;
        let placed = |placement: StartupPlacement| -> Vec<ContextProvenance> {
            rows.iter()
                .filter(|row| row.startup_placement == placement)
                .cloned()
                .collect()
        };
        let system_rows = placed(StartupPlacement::System);
        let user_rows = placed(StartupPlacement::User);
        let missing = request
            .system_modules
            .iter()
            .zip(std::iter::repeat(&system_rows))
            .chain(
                request
                    .user_modules
                    .iter()
                    .zip(std::iter::repeat(&user_rows)),
            )
            .find_map(|(identity, placed)| {
                (!placed.iter().any(|row| &row.module_identity == identity))
                    .then(|| identity.clone())
            });
        if let Some(identity) = missing {
            return Err(ContextRefusal::ModuleMissing(Some(identity)));
        }
        let ordered = |placed: &[ContextProvenance]| {
            placed
                .iter()
                .enumerate()
                .all(|(ordinal, row)| row.integer == ordinal as i64)
        };
        if !ordered(&system_rows) || !ordered(&user_rows) {
            return Err(ContextRefusal::PayloadMismatch);
        }
        if composition
            .system_text
            .rendered_segments(&system_rows)
            .is_none()
        {
            return Err(ContextRefusal::PayloadMismatch);
        }
        let user_segments = composition
            .user_text
            .rendered_segments(&user_rows)
            .ok_or(ContextRefusal::PayloadMismatch)?;
        Ok(Self {
            address: request.address.clone(),
            system_text: composition.system_text,
            user_segments,
            records_text: composition.records_text,
        })
    }
}

  directory="$1"
  url="$2"
  if [ -d "$directory/.git" ]; then
    actual_url=$("$git" -C "$directory" config --get remote.origin.url)
    if [ "$actual_url" != "$url" ]; then
      echo "unexpected context origin for $directory" >&2
      exit 1
    fi
    if [ -n "$("$git" -C "$directory" status --porcelain)" ]; then
      echo "refusing to discard local context changes in $directory" >&2
      exit 1
    fi
    "$git" -C "$directory" fetch --quiet --no-tags --prune origin \
      +refs/heads/main:refs/remotes/origin/main
    "$git" -C "$directory" checkout --quiet --force --detach \
      refs/remotes/origin/main
  else
    if [ -e "$directory" ] || [ -L "$directory" ]; then
      echo "context path is not a managed checkout: $directory" >&2
      exit 1
    fi
    mkdir -p "$(dirname "$directory")"
    "$git" clone --quiet --depth 1 --single-branch \
      --branch main "$url" "$directory"
  fi
  local_revision=$("$git" -C "$directory" rev-parse HEAD)
  remote_revision=$("$git" -C "$directory" \
    rev-parse refs/remotes/origin/main)
  if [ "$local_revision" != "$remote_revision" ]; then
    echo "context checkout is not origin/main: $directory" >&2
    exit 1
  fi
}
